# ==================== 2. SMART TAR ARCHIVE UPLOADER (ZERO FREEZING) ====================
def hf_upload_preprocessed_tar(local_folder, repo_folder):
    tar_path = os.path.join(KAGGLE_WORKING, "preprocessed_data.tar.gz")
    print(f"📦 Compressing {local_folder} into {tar_path} for ultra-fast, 100% stable upload...")
    with tarfile.open(tar_path, "w:gz") as tar:
        tar.add(local_folder, arcname="preprocessed_data")
    
    file_size_gb = os.path.getsize(tar_path) / (1024**3)
    print(f"🚀 Uploading single archive ({file_size_gb:.2f} GB) to HF/{repo_folder}...")
    
    hf_api.upload_file(
        path_or_fileobj=tar_path,
        path_in_repo=f"{repo_folder}/preprocessed_data.tar.gz",
        repo_id=HF_BACKUP_REPO,
        repo_type="model",
    )
    print("✅ Archive uploaded to Hugging Face successfully!")
    if os.path.exists(tar_path):
        os.remove(tar_path)
